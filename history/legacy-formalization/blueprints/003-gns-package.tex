\begin{theorem}[GNSPackage, superseded ID: GNS-001]
Let \(A\) be a unital \(C^*\)-algebra and let \(\omega\) be a state on \(A\).
There exists a Hilbert space \(H_\omega\), a unital \(*\)-representation
\[
  \pi_\omega:A\to B(H_\omega),
\]
and a cyclic unit vector \(\Omega_\omega\in H_\omega\) such that
\[
  \omega(a)=\langle \Omega_\omega,\pi_\omega(a)\Omega_\omega\rangle
\]
for all \(a\in A\).
\source{Kadison--Ringrose, \emph{Fundamentals of the Theory of Operator
Algebras, Vol.~I}, Theorem 4.5.2 and Proposition 4.5.3, PDF pp.~287--288.
Lean infrastructure: \texttt{Mathlib.Analysis.CStarAlgebra.GelfandNaimarkSegal};
see also the coverage audit in
\texttt{agents/2026-06-21-mathlib-operator-algebra-coverage-audit.md}.}
\lean{Formalization.GNS.GNSPackage}

\end{theorem}
